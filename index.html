CREATE TABLE cortes_fab (id text primary key, numero text, descricao text, quantidade int, data_corte date, obs text, created_at bigint);
CREATE TABLE estamparia_fab (id text primary key, corte_id text, numero_corte text, descricao text, estamparia_nome text, data_envio date, entrega_total int, devolucao_total int, status text, valor_unit numeric, valor_total numeric, pago boolean default false, obs text, created_at bigint);
CREATE TABLE costureiro_fab (id text primary key, corte_id text, numero_corte text, descricao text, oficina_nome text, data_envio date, quantidade int, parcial int, total int, devolucao_total int, status text, valor_unit numeric, valor_total numeric, valor_pago numeric default 0, obs text, created_at bigint);
CREATE TABLE despesas_fab (id text primary key, data date, descricao text, valor numeric, responsavel text, created_at bigint);
